Pseudo-3D Road

A road that curves back and forth, like an old racing game, made with the ThreeDee Module.

Steer with the left and right arrow keys.

The Code

Make a project like in Getting Started, and put this in game.js. It doesn't need any assets.

import MDog from "https://cdn.jsdelivr.net/gh/MiloKesteloot/MDogEngine@v1.4.0/MDogModules/MDogMain.js";

const ThreeDee = MDog.ThreeDee;
const Vector3 = MDog.Math.Vector3;
const Keyboard = MDog.Input.Keyboard;

// The ground is 10 below the camera
const groundY = 10;
const camera = new ThreeDee.Camera(0, 0, 0, 100);
const stripeLength = 4;
const viewDistance = 80;

function main() {
    // Drive forward, and steer with the arrow keys
    camera.position.z += 0.15;
    if (Keyboard.isDown("ArrowLeft")) {
        camera.position.x -= 0.1;
    }
    if (Keyboard.isDown("ArrowRight")) {
        camera.position.x += 0.1;
    }

    // A new scene every tick, with just the stripes in view. The closest stripe gets shorter as the camera
    // drives over it, and the farthest ones have to be added first, so it's simplest to start over every tick.
    const scene = new ThreeDee.ThreeDeeScene(camera);

    // Stripes are added from farthest to closest, so the closer ones are drawn on top.
    // The closest edge stays a little in front of the camera, since points at the camera's z can't be drawn.
    const near = camera.getZ() + 3;
    const far = camera.getZ() + viewDistance;
    for (let z = Math.floor(far / stripeLength) * stripeLength; z + stripeLength > near; z -= stripeLength) {
        const zFar = z + stripeLength;
        const zNear = Math.max(z, near);
        const light = Math.floor(z / stripeLength) % 2 === 0;

        // Grass
        scene.addObject(new ThreeDee.ThreeDeeFloor(groundY, zFar, zNear, light ? "#4a8f42" : "#3f7d3a"));

        // Road
        scene.addObject(new ThreeDee.ThreeDeeShape([
            new Vector3(roadCenter(zNear) - 6, groundY, zNear),
            new Vector3(roadCenter(zNear) + 6, groundY, zNear),
            new Vector3(roadCenter(zFar) + 6, groundY, zFar),
            new Vector3(roadCenter(zFar) - 6, groundY, zFar),
        ], light ? "#666666" : "#5a5a5a"));
    }

    MDog.Draw.clear({color: "#87b8e8"});
    MDog.Draw.threeDeeScene(scene);
}

// The road curves back and forth
function roadCenter(z) {
    return Math.sin(z / 40) * 12;
}

MDog.setActiveFunction(main);

How it Works

The camera drives forward by adding to camera.position.z every tick, and steers by changing camera.position.x.

The ground is split into stripes that alternate between two colors, and that's what makes it look like it's moving. Each stripe is a ThreeDeeFloor for the grass and a ThreeDeeShape for the road.

We make a brand new scene every tick with only the stripes in front of the camera. Things are drawn in the order they're added, so stripes go in from farthest to closest. The closest stripe gets cut shorter as you drive over it, so rebuilding every tick is a lot simpler than shuffling stripes around with removeObject().

The middle of the road wiggles side to side with Math.sin(), and that's the whole curve.
