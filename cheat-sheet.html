MDog.Draw.setScreenSize(256, 192);  // Optional, 512 by 384 if not

function main() {
    // Runs 160 times per second
}

MDog.setActiveFunction(main);

Drawing

MDog.Draw.clear({color: "#1c1c3c"});
MDog.Draw.rectangleFill(x, y, width, height, "#ff8800");

MDog.Draw.line(x1, y1, x2, y2, "#88ccff");
MDog.Draw.point(x, y, "#ffffff");

Text

Animations

Keyboard

const Keyboard = MDog.Input.Keyboard;
Keyboard.isDown("ArrowLeft")    // Held down
Keyboard.isClicked(" ")         // Just pressed (space)
// Keys: "a", "1", " ", "ArrowUp", "Shift", "Enter", "Escape"

Mouse

const Mouse = MDog.Input.Mouse;
Mouse.getX(), Mouse.getY()   // In art pixels
Mouse.getClick()             // Just pressed
Mouse.getDown()              // Held down
Mouse.getClick(2)            // Right button

Layers and the Camera

MDog.Draw.circle(x, y, 5, "#fff", {layer: 1});  // On top of layer 0
MDog.Draw.translate(-cameraX, 0);  // Shift what's drawn next
MDog.Draw.translate(0, 0);         // Back to normal

Particles

const sparks = new MDog.FX.ParticleSystem();
sparks.addParticle(new MDog.FX.ChunkParticle(x, y, 60, "#ffcc00", vx, vy, {gy: 0.05}));
sparks.update();                         // Every tick
MDog.Draw.particleSystem(sparks);
