Examples

Title Screen

A title screen that starts the game when you press space, and a game that goes back to the title when you press Escape. It shows how to switch screens with setActiveFunction().

The Code

Make a project like in Getting Started, and put this in game.js. It uses the marsfont.png font, so put that in assets/fonts/ (see Add Text).

import MDog from "https://cdn.jsdelivr.net/gh/MiloKesteloot/MDogEngine@v1.4.0/MDogModules/MDogMain.js";

const font = "fonts/marsfont.png";
const Keyboard = MDog.Input.Keyboard;

let ticks = 0;
const ball = new MDog.Math.Vector();
const ballVelocity = new MDog.Math.Vector();

function titleScreen() {
    ticks++;

    MDog.Draw.clear({color: "#1a1030"});
    MDog.Draw.textImage("Bouncy Ball", 256, 140, "#ffcc33", font, {alignX: "center", size: 5});

    // Blink on and off about twice a second
    if (Math.floor(ticks / 40) % 2 === 0) {
        MDog.Draw.textImage("Press space to start", 256, 230, "#ffffff", font, {alignX: "center", size: 2});
    }

    if (Keyboard.isClicked(" ")) {
        startGame();
    }
}

function startGame() {
    ball.set(256, 192);
    ballVelocity.set(1.2, 0.9);
    MDog.setActiveFunction(game);
}

function game() {
    ball.add(ballVelocity);

    // Bounce off the edges of the screen
    if (ball.x < 6 || ball.x > MDog.Draw.getScreenWidthInArtPixels() - 6) {
        ballVelocity.x *= -1;
    }
    if (ball.y < 6 || ball.y > MDog.Draw.getScreenHeightInArtPixels() - 6) {
        ballVelocity.y *= -1;
    }

    MDog.Draw.clear({color: "#102030"});
    MDog.Draw.circle(Math.floor(ball.x), Math.floor(ball.y), 6, "#ff5566");
    MDog.Draw.textImage("Escape to go back", 8, 8, "#ffffff", font);

    if (Keyboard.isClicked("Escape")) {
        MDog.setActiveFunction(titleScreen);
    }
}

MDog.setActiveFunction(titleScreen);

How it Works

Each screen is its own function. MDog.setActiveFunction() changes which one runs every tick, so switching screens is one line.

startGame() puts the ball back in the middle before switching, so you get a fresh game every time.

The blinking text just counts ticks. Math.floor(ticks / 40) % 2 flips between 0 and 1 every 40 ticks, which is a quarter of a second.
