Examples

Particles

Explosions and sparkles made with the FX Module.

Click to make an explosion, and hold the mouse down to leave a trail of sparkles.

The Code

Make a project like in Getting Started, and put this in game.js. It doesn't need any assets.

import MDog from "https://cdn.jsdelivr.net/gh/MiloKesteloot/MDogEngine@v1.1.0/MDogModules/MDogMain.js";

const Mouse = MDog.Input.Mouse;
const particles = new MDog.FX.ParticleSystem();
const colors = ["#ffcc33", "#ff8833", "#ff4455"];

function main() {
    // Click to make an explosion
    if (Mouse.getClick()) {
        explode(Mouse.getX(), Mouse.getY());
    }

    // Hold the mouse down to leave a trail of sparkles
    if (Mouse.getDown() && Math.random() < 0.3) {
        particles.addParticle(new MDog.FX.ChunkParticle(
            Mouse.getX(), Mouse.getY(), 80, "#88ccff", random(-0.3, 0.3), random(-0.6, -0.2), {gy: 0.005}));
    }

    particles.update();

    MDog.Draw.clear({color: "#101018"});
    MDog.Draw.particleSystem(particles);
}

function explode(x, y) {
    // Chunks that fly out and fall
    for (let i = 0; i < 40; i++) {
        // A random direction, made by turning a vector by a random angle
        const velocity = new MDog.Math.Vector(random(0.5, 2), 0).rotate(random(0, 360));
        const color = colors[Math.floor(Math.random() * colors.length)];
        const life = Math.floor(random(60, 140));
        particles.addParticle(new MDog.FX.ChunkParticle(x, y, life, color, velocity.x, velocity.y, {gy: 0.02, size: 2}));
    }

    // Fast white streaks
    for (let i = 0; i < 12; i++) {
        const velocity = new MDog.Math.Vector(random(2, 3), 0).rotate(random(0, 360));
        particles.addParticle(new MDog.FX.LineParticle(x, y, 30, "#ffffff", velocity.x, velocity.y, {length: 6}));
    }
}

function random(min, max) {
    return min + Math.random() * (max - min);
}

MDog.setActiveFunction(main);

How it Works

All of the particles are in one ParticleSystem. Every tick, ParticleSystem.update() moves the particles and removes the ones whose life ran out, then MDog.Draw.particleSystem() draws them.

Each explosion is made of two kinds of particles: ChunkParticles that fly out and fall with gravity, and LineParticles that streak outward.

The random directions come from making a Vector that points right, then turning it a random amount with rotate().
